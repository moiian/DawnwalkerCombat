cmake_minimum_required(VERSION 3.28)
project(DawnwalkerCombat VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(DW_CORE_TESTS_ONLY "Build portable input tests without SKSE" OFF)
enable_testing()
add_executable(DirectionTests tests/DirectionTests.cpp)
target_include_directories(DirectionTests PRIVATE src)
add_test(NAME DirectionTests COMMAND DirectionTests)
if(DW_CORE_TESTS_ONLY)
  return()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The plugin requires Windows x64")
endif()

set(BUILD_TESTS OFF CACHE BOOL "CommonLib internal tests" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(CommonLibSSE
  GIT_REPOSITORY https://github.com/CharmedBaryon/CommonLibSSE-NG.git
  GIT_TAG v3.7.0
  GIT_SHALLOW TRUE
  EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(CommonLibSSE)

add_library(DawnwalkerCombat SHARED src/Plugin.cpp src/Direction.h)
target_link_libraries(DawnwalkerCombat PRIVATE CommonLibSSE::CommonLibSSE user32 comctl32)
target_compile_definitions(DawnwalkerCombat PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(DawnwalkerCombat PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_precompile_headers(DawnwalkerCombat PRIVATE <RE/Skyrim.h> <SKSE/SKSE.h>)
install(TARGETS DawnwalkerCombat RUNTIME DESTINATION SKSE/Plugins)
install(DIRECTORY package/ DESTINATION .)
install(FILES README.md LICENSE.md THIRD_PARTY_NOTICES.md DESTINATION docs/DawnwalkerCombat)
install(FILES "${commonlibsse_SOURCE_DIR}/LICENSE" DESTINATION docs/DawnwalkerCombat/licenses RENAME CommonLibSSE-NG.txt)
install(DIRECTORY "${VCPKG_INSTALLED_DIR}/x64-windows-static/share/fmt/" DESTINATION docs/DawnwalkerCombat/licenses/fmt FILES_MATCHING PATTERN copyright)
install(DIRECTORY "${VCPKG_INSTALLED_DIR}/x64-windows-static/share/spdlog/" DESTINATION docs/DawnwalkerCombat/licenses/spdlog FILES_MATCHING PATTERN copyright)
install(DIRECTORY "${VCPKG_INSTALLED_DIR}/x64-windows-static/share/rapidcsv/" DESTINATION docs/DawnwalkerCombat/licenses/rapidcsv FILES_MATCHING PATTERN copyright)
